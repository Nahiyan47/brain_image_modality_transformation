def center_crop(img, size):
    h, w = img.shape[-2:]  # Get height and width
    th, tw = size          # Target height and width
    i = (h - th) // 2
    j = (w - tw) // 2
    return img[..., i:i+th, j:j+tw]  # Crop the central region

# Define Generator and Discriminator

class UNetGenerator(nn.Module):
    def __init__(self, in_channels=1, out_channels=1):
        super(UNetGenerator, self).__init__()
        
        # Encoder
        self.enc1 = self.conv_block(in_channels, 64)
        self.enc2 = self.conv_block(64, 128)
        self.enc3 = self.conv_block(128, 256)
        self.enc4 = self.conv_block(256, 512)
        
        # Bottleneck
        self.bottleneck = self.conv_block(512, 1024)
        
        # Decoder
        self.dec4 = self.upconv_block(1024, 512)
        self.dec3 = self.upconv_block(512, 256)
        self.dec2 = self.upconv_block(256, 128)
        self.dec1 = self.upconv_block(128, 64)
        
        # Final Convolution
        self.final_conv = nn.Conv2d(64, out_channels, kernel_size=1)

        # Intermediate convolution layers to reduce channels after concatenation
        self.reduce4 = self.conv_block(1024, 512)  # After concatenation of dec4 and enc4
        self.reduce3 = self.conv_block(512, 256)   # After concatenation of dec3 and enc3
        self.reduce2 = self.conv_block(256, 128)   # After concatenation of dec2 and enc2
        self.reduce1 = self.conv_block(128, 64)    # After concatenation of dec1 and enc1

    def conv_block(self, in_channels, out_channels):
        return nn.Sequential(
            nn.Conv2d(in_channels, out_channels, kernel_size=3, padding=1),
            nn.BatchNorm2d(out_channels),
            nn.ReLU(inplace=True),
            nn.Conv2d(out_channels, out_channels, kernel_size=3, padding=1),
            nn.BatchNorm2d(out_channels),
            nn.ReLU(inplace=True)
        )

    def upconv_block(self, in_channels, out_channels):
        return nn.Sequential(
            nn.ConvTranspose2d(in_channels, out_channels, kernel_size=2, stride=2),
            nn.ReLU(inplace=True)
        )

    def forward(self, x):
        # Encoder path (downsampling)
        enc1 = self.enc1(x)
        enc2 = self.enc2(nn.functional.max_pool2d(enc1, 2))
        enc3 = self.enc3(nn.functional.max_pool2d(enc2, 2))
        enc4 = self.enc4(nn.functional.max_pool2d(enc3, 2))
        
        # Bottleneck (deepest layer)
        bottleneck = self.bottleneck(nn.functional.max_pool2d(enc4, 2))
        
        # Decoder path with skip connections
        dec4 = self.dec4(bottleneck)
        # Resize dec4 to match enc4 dimensions before concatenation
        dec4 = nn.functional.interpolate(dec4, size=enc4.shape[2:], mode='bilinear', align_corners=False)
        dec4 = torch.cat((dec4, enc4), dim=1)  # Skip connection
        dec4 = self.reduce4(dec4)  # Reduce channels after concatenation
    
        dec3 = self.dec3(dec4)
        # Resize dec3 to match enc3 dimensions before concatenation
        dec3 = nn.functional.interpolate(dec3, size=enc3.shape[2:], mode='bilinear', align_corners=False)
        dec3 = torch.cat((dec3, enc3), dim=1)  # Skip connection
        dec3 = self.reduce3(dec3)  # Reduce channels after concatenation
    
        dec2 = self.dec2(dec3)
        # Resize dec2 to match enc2 dimensions before concatenation
        dec2 = nn.functional.interpolate(dec2, size=enc2.shape[2:], mode='bilinear', align_corners=False)
        dec2 = torch.cat((dec2, enc2), dim=1)  # Skip connection
        dec2 = self.reduce2(dec2)  # Reduce channels after concatenation
    
        dec1 = self.dec1(dec2)
        # Resize dec1 to match enc1 dimensions before concatenation
        dec1 = nn.functional.interpolate(dec1, size=enc1.shape[2:], mode='bilinear', align_corners=False)
        dec1 = torch.cat((dec1, enc1), dim=1)  # Skip connection
        dec1 = self.reduce1(dec1)  # Reduce channels after concatenation
        
        # Final output layer: Apply a sigmoid activation for output in range [0, 1]
        return torch.sigmoid(self.final_conv(dec1))



class Discriminator(nn.Module):
    def __init__(self):
        super(Discriminator, self).__init__()
        # Define layers (example architecture)
        self.conv1 = nn.Conv2d(1, 64, kernel_size=3, padding=1)
        self.relu = nn.ReLU()
        self.conv2 = nn.Conv2d(64, 1, kernel_size=3, padding=1)  # Output channel 1

    def forward(self, x):
        x = self.relu(self.conv1(x))
        x = torch.sigmoid(self.conv2(x))  # Sigmoid activation to constrain output
        return x
